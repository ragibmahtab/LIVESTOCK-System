-- Director (Production) — view and procedure



/// created view for director production profile info

CREATE OR REPLACE VIEW Director_Production_Profile AS
SELECT
    ui.User_ID,
    ui.Name,
    ui.Username,
    ui.Email,
    dp.Gradation_No,
    dp.Farm_Type,
    TO_CHAR(dp.Appointment_Date, 'DD Month, YYYY') AS Appointment_Date,
    LISTAGG(up.Phone, ', ') WITHIN GROUP (ORDER BY up.Phone) AS Phones
FROM USER_INFO ui
JOIN Director_Production dp ON ui.User_ID = dp.Director_Production_ID
JOIN User_Phone up ON up.User_ID = ui.User_ID
GROUP BY ui.User_ID, ui.Name, ui.Username, ui.Email,
         dp.Gradation_No, dp.Farm_Type, dp.Appointment_Date
WITH READ ONLY CONSTRAINT Director_Production_Profile_RO;


/// created a procedure to create farm supply with user defined exceptions

CREATE OR REPLACE PROCEDURE CREATE_FARM_SUPPLY (
    p_director_id  IN  NUMBER,
    p_demand_id    IN  VARCHAR2,
    p_quantity     IN  NUMBER,
    p_cost         IN  NUMBER,
    p_supply_date  IN  VARCHAR2,   
    p_supply_id    OUT VARCHAR2
)
AS
    
    demand_not_approved  EXCEPTION;
    already_supplied     EXCEPTION;
    quantity_too_high    EXCEPTION;

    v_status          Farm_Demand.Status%TYPE;
    v_requested       Farm_Demand.Requested_Quantity%TYPE;
    v_farm_manager_id Farm_Demand.Creator_ID%TYPE;
    v_count           NUMBER;
BEGIN

    
    SELECT Status, Requested_Quantity, Creator_ID
    INTO   v_status, v_requested, v_farm_manager_id
    FROM   Farm_Demand
    WHERE  Farm_Demand_ID = p_demand_id
    AND    Approved_By_ID = p_director_id;

    IF v_status <> 'Approved' THEN
        RAISE demand_not_approved;
    END IF;

    SELECT COUNT(*) INTO v_count
    FROM   Farm_Supply
    WHERE  Farm_Demand_ID = p_demand_id;

    IF v_count > 0 THEN
        RAISE already_supplied;
    END IF;

    IF p_quantity > v_requested THEN
        RAISE quantity_too_high;
    END IF;

    
    SELECT 'FS' || LPAD(NVL(MAX(TO_NUMBER(SUBSTR(Farm_Supply_ID, 3))), 0) + 1, 3, '0')
    INTO   p_supply_id
    FROM   Farm_Supply;

    INSERT INTO Farm_Supply
        (Farm_Supply_ID, Granted_Quantity, Cost, Creator_ID, Farm_Demand_ID, Farm_Manager_ID, Supply_Date)
    VALUES
        (p_supply_id, p_quantity, p_cost, p_director_id, p_demand_id, v_farm_manager_id,
         TO_DATE(p_supply_date, 'YYYY-MM-DD'));

    COMMIT;

EXCEPTION
      WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20011, 'This demand request does not belong to this director.');
    WHEN demand_not_approved THEN
        RAISE_APPLICATION_ERROR(-20012, 'Supply can only be created for an approved demand.');
    WHEN already_supplied THEN
        RAISE_APPLICATION_ERROR(-20013, 'A supply already exists for this demand.');
    WHEN quantity_too_high THEN
        RAISE_APPLICATION_ERROR(-20014, 'Granted quantity cannot be more than the requested quantity.');
END;
/


-- farm_manager_low_stock_proc

CREATE OR REPLACE PROCEDURE get_farm_low_stock (
    p_manager_id  IN  NUMBER,
    p_result      OUT SYS_REFCURSOR
)
IS
    -- The farm this manager is responsible for
    v_farm_id  Farm_Manager.Assigned_Farm%TYPE;

    -- Explicit cursor over every resource on that farm
    CURSOR c_resources IS
        SELECT Resource_ID, Name, Current_Quantity, Minimum_Quantity
        FROM Resources
        WHERE Farm_ID = v_farm_id;

    v_row       c_resources%ROWTYPE;
    v_low_count NUMBER := 0;

    
    e_no_farm EXCEPTION;

BEGIN

    SELECT Assigned_Farm
    INTO v_farm_id
    FROM Farm_Manager
    WHERE Manager_ID = p_manager_id;

    IF v_farm_id IS NULL THEN
        RAISE e_no_farm;
    END IF;

   
    OPEN c_resources;

    LOOP
        FETCH c_resources INTO v_row;
        EXIT WHEN c_resources%NOTFOUND;

        IF v_row.Current_Quantity < v_row.Minimum_Quantity THEN
            v_low_count := v_low_count + 1;
            DBMS_OUTPUT.PUT_LINE(v_row.Name || ' is low: '
                || v_row.Current_Quantity || ' of ' || v_row.Minimum_Quantity);
        END IF;
    END LOOP;

    CLOSE c_resources;

    DBMS_OUTPUT.PUT_LINE('Total low stock resources: ' || v_low_count);

       OPEN p_result FOR
        SELECT Resource_ID, Name, Current_Quantity, Minimum_Quantity
        FROM Resources
        WHERE Farm_ID = v_farm_id
        AND Current_Quantity < Minimum_Quantity
        ORDER BY Current_Quantity;

EXCEPTION

    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20001, 'No farm manager found with that ID.');

    WHEN e_no_farm THEN
        RAISE_APPLICATION_ERROR(-20002, 'This farm manager has no farm assigned yet.');

END;
/
