CREATE TABLE USER_INFO (
    User_ID     INT CONSTRAINT USER_INFO_USER_ID_PK PRIMARY KEY,
    Username    VARCHAR(50) CONSTRAINT USER_INFO_USERNAME_NN NOT NULL
                            CONSTRAINT USER_INFO_USERNAME_UK UNIQUE,
    Password    VARCHAR(100) CONSTRAINT USER_INFO_PASSWORD_NN NOT NULL,
    Name        VARCHAR(100) CONSTRAINT USER_INFO_NAME_NN NOT NULL,
    Email       VARCHAR(100) CONSTRAINT USER_INFO_EMAIL_UK UNIQUE
);

CREATE TABLE User_Phone (
    User_ID     INT CONSTRAINT USER_PHONE_USER_ID_NN NOT NULL,
    Phone       VARCHAR(20) CONSTRAINT USER_PHONE_PHONE_NN NOT NULL,
    CONSTRAINT USER_PHONE_PK PRIMARY KEY (User_ID, Phone),
    CONSTRAINT USER_PHONE_USER_ID_FK FOREIGN KEY (User_ID) REFERENCES USER_INFO(User_ID) ON DELETE CASCADE
);

CREATE TABLE District_Office (
    Dist_Off_ID INT CONSTRAINT DISTRICT_OFFICE_DIST_OFF_ID_PK PRIMARY KEY,
    Dist_Name   VARCHAR(50) CONSTRAINT DISTRICT_OFFICE_DIST_NAME_NN NOT NULL,
    Dist_ID     VARCHAR(10) CONSTRAINT DISTRICT_OFFICE_DIST_ID_NN NOT NULL 
                            CONSTRAINT DISTRICT_OFFICE_DIST_ID_UK UNIQUE,
    CONSTRAINT DISTRICT_OFFICE_DIST_OFF_ID_FK FOREIGN KEY (Dist_Off_ID) REFERENCES USER_INFO(User_ID) ON DELETE CASCADE
);


CREATE TABLE Upazila_Office (
    Off_ID      INT CONSTRAINT UPAZILA_OFFICE_OFF_ID_PK PRIMARY KEY,
    Upz_Name    VARCHAR(50) CONSTRAINT UPAZILA_OFFICE_UPZ_NAME_NN NOT NULL,
    Upz_ID      VARCHAR(10) CONSTRAINT UPAZILA_OFFICE_UPZ_ID_NN NOT NULL 
                            CONSTRAINT UPAZILA_OFFICE_UPZ_ID_UK UNIQUE,
    Dist_ID     VARCHAR(10) CONSTRAINT UPAZILA_OFFICE_DIST_ID_NN NOT NULL,
    CONSTRAINT UPAZILA_OFFICE_OFF_ID_FK FOREIGN KEY (Off_ID) REFERENCES USER_INFO(User_ID) ON DELETE CASCADE,
    CONSTRAINT UPAZILA_OFFICE_DIST_ID_FK FOREIGN KEY (Dist_ID) REFERENCES District_Office(Dist_ID)
